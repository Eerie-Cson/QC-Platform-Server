// 1. Reusable delay helper for async/await
const delay = (ms) => new Promise(resolve => setTimeout(resolve, ms));

const video = document.querySelector('video');

(async () => {
    if (!video) {
        console.error("❌ No video element found on this page.");
        return;
    }

    console.log("⏳ Initialized. Waiting 10 seconds before starting loop...");
    await delay(10000); // 10-second delay at the start

    console.log("Loop active. 🛑 RIGHT-CLICK to DESTROY. ⏸️ MIDDLE-CLICK to PAUSE/RESUME.");
    
    let isWaitingForBuffer = false;
    let isPausedManually = false; 

    // 2. The Automation Loop
    const arrowLoop = setInterval(() => {
        if (video.ended) {
            cleanup();
            console.log("Video finished! Loop cleared.");
            return;
        }

        if (video.buffered.length > 0) {
            const currentBufferEnd = video.buffered.end(video.buffered.length - 1);
            const bufferAhead = currentBufferEnd - video.currentTime;

            if (isWaitingForBuffer) {
                // DYNAMIC RESUME: Resume if the gray bar has built up to 35+ seconds ahead
                if (bufferAhead >= 35) {
                    isWaitingForBuffer = false;
                    console.log(`▶️ Gray bar is now ${bufferAhead.toFixed(1)}s ahead. Resuming automation.`);
                }
            } else {
                // TRIGGER PAUSE: Pause if the white bar gets within 25 seconds of the gray bar's end
                if (bufferAhead < 25) {
                    isWaitingForBuffer = true;
                    console.log(`⏳ Buffer low (${bufferAhead.toFixed(1)}s ahead). Waiting for gray bar to load...`);
                }
            }
        }

        // FIRE KEY EVENT: Only runs if conditions are optimal
        if (!video.paused && !video.seeking && !isWaitingForBuffer && !isPausedManually) {
            video.focus();
            const rightArrowEvent = new KeyboardEvent('keydown', { 
                key: 'ArrowRight', 
                code: 'ArrowRight', 
                keyCode: 39, 
                bubbles: true, 
                cancelable: true 
            });
            video.dispatchEvent(rightArrowEvent);
        }
    }, 600);

    // 3. Mouse Trigger Event Handler
    const handleMouseClicks = (event) => {
        // MIDDLE CLICK (button 1) or SIDE BUTTONS (button 3, button 4)
        if (event.button === 1 || event.button === 3 || event.button === 4) {
            event.preventDefault(); // Stop default middle-click scroll behavior
            isPausedManually = !isPausedManually;
            console.log(isPausedManually ? "⏸️ Automation manually PAUSED." : "▶️ Automation manually RESUMED.");
        }
        
        // RIGHT CLICK (button 2)
        if (event.button === 2) {
            event.preventDefault(); // Stop context menu from showing up
            cleanup();
            console.log("🛑 Automation DESTROYED and event listeners removed.");
        }
    };

    // 4. Cleanup Function to wipe everything out safely
    const cleanup = () => {
        clearInterval(arrowLoop);
        window.removeEventListener('mousedown', handleMouseClicks);
    };

    // 5. Attach the mouse listener to the global window
    window.addEventListener('mousedown', handleMouseClicks);

})();
