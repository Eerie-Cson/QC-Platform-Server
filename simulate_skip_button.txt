const video = document.querySelector('video');

if (video) {
  console.log('Loop active. 🛑 RIGHT-CLICK to DESTROY. ⏸️ MIDDLE-CLICK (or mouse side buttons) to PAUSE/RESUME.');

  let isWaitingForBuffer = false;
  let isPausedManually = false; // Tracks if user paused it via mouse button

  const arrowLoop = setInterval(() => {
    if (video.ended) {
      clearInterval(arrowLoop);
      console.log('Video finished! Loop cleared.');
      return;
    }

    if (video.buffered.length > 0) {
      const currentBufferEnd = video.buffered.end(video.buffered.length - 1);
      const bufferAhead = currentBufferEnd - video.currentTime;

      if (isWaitingForBuffer) {
        // DYNAMIC RESUME: Only resume if the gray bar has built up to 15+ seconds ahead
        if (bufferAhead >= 35) {
          isWaitingForBuffer = false;
          console.log(`▶️ Gray bar is now ${bufferAhead.toFixed(1)}s ahead. Resuming automation.`);
        }
      } else {
        // TRIGGER PAUSE: Pause if the white bar gets within 15 seconds of the gray bar's end
        if (bufferAhead < 25) {
          isWaitingForBuffer = true;
          console.log(`⏳ Buffer low (${bufferAhead.toFixed(1)}s ahead). Waiting for gray bar to load...`);
        }
      }
    }

    // FIRE KEY EVENT: Only runs if the video is playing, buffer is healthy, AND not manually paused
    if (!video.paused && !video.seeking && !isWaitingForBuffer && !isPausedManually) {
      video.focus();
      
      const rightArrowEvent = new KeyboardEvent('keydown', {
        key: 'ArrowRight',
        code: 'ArrowRight',
        keyCode: 39,
        bubbles: true,
        cancelable: true
      });

      video.dispatchEvent(rightArrowEvent);
    }
  }, 600);

  // MOUSE TRIGGER FOR PAUSE: Middle-click or Side buttons
  const handleMousePause = (event) => {
    // button 1 = Middle Click, button 3 = Back Side Button, button 4 = Forward Side Button
    if (event.button === 1 || event.button === 3 || event.button === 4) {
      event.preventDefault(); // Prevents default browser actions for these buttons
      isPausedManually = !isPausedManually; // Toggle the state
      
      if (isPausedManually) {
        console.log('⏸️ Automation PAUSED manually. Click Middle or Side mouse button to resume.');
      } else {
        console.log('▶️ Automation RESUMED.');
      }
    }
  };

  // KILL SWITCH: Right-click cleanly destroys everything
  const stopScript = (event) => {
    event.preventDefault(); // Prevents the right-click menu from opening
    clearInterval(arrowLoop); // Stops the loop
    document.removeEventListener('contextmenu', stopScript); // Cleans up right-click listener
    document.removeEventListener('mousedown', handleMousePause); // Cleans up pause listener
    console.log('🛑 Automation stopped manually by right-click!');
  };

  // Listen for the clicks
  document.addEventListener('contextmenu', stopScript);
  document.addEventListener('mousedown', handleMousePause); // mousedown catches auxiliary clicks

} else {
  console.log('Video element not found.');
}
